# 核心数据模型设计

## 1. 装备（Item）

```typescript
// 装备品质
enum Rarity {
  Normal = "normal",       // 白色 - 无词缀
  Magic = "magic",         // 蓝色 - 最多3前缀+3后缀（实际常见1-2）
  Rare = "rare",           // 黄色 - 最多3前缀+3后缀
  Unique = "unique"        // 橙色 - 固定词缀组合，不可改造
}

// 装备部位
enum EquipSlot {
  Weapon = "weapon",       // 主手武器
  Offhand = "offhand",     // 副手（盾/副手武器）
  Helmet = "helmet",
  Body = "body",           // 胸甲
  Gloves = "gloves",
  Boots = "boots",
  Belt = "belt",
  Amulet = "amulet",
  Ring1 = "ring1",
  Ring2 = "ring2"
}

// 宝石插槽
interface Socket {
  color: "red" | "green" | "blue" | "white";
  gemId: string | null;    // 当前镶嵌的宝石ID
  linkedTo: number[];      // 连接到哪些插槽（by index）
}

// 一件装备
interface Item {
  id: string;
  name: string;            // 基底名称 + 可能的稀有名称
  type: ItemType;          // 子类型（长剑/弓/胸甲等）
  slot: EquipSlot;
  rarity: Rarity;
  itemLevel: number;       // 物品等级，决定可出现的词缀等级
  base: BaseItem;          // 基底数据引用
  
  // 基底固有属性（Implicit）
  implicit: Affix[];
  
  // 随机词缀
  prefixes: Affix[];       // 前缀，最多3个
  suffixes: Affix[];       // 后缀，最多3个
  
  // 插槽
  sockets: Socket[];
  
  // 品质
  quality: number;         // 0-20
  
  // 限定词缀（Unique专用）
  isFixed?: boolean;       // true则不可被改造通货改变
}
```

## 2. 词缀（Affix）

```typescript
// 词缀类别
enum AffixCategory {
  Prefix = "prefix",
  Suffix = "suffix"
}

// 词缀标签（用于限制可出现的装备部位）
enum AffixTag {
  Weapon = "weapon",
  Armor = "armor",
  Jewelry = "jewelry",
  // ... 等等
}

// 单条词缀
interface Affix {
  id: string;              // 唯一标识，如 "flat_life"
  name: string;            // 显示名，如 "鲜活的"
  category: AffixCategory;
  tags: AffixTag[];        // 可出现在哪些装备上
  tier: number;            // 词缀等级 (1最高, 越大越弱)
  itemLevelReq: number;    // 装备物品等级要求
  
  // 属性加成（支持多属性）
  stats: StatBonus[];
}

// 属性加成
interface StatBonus {
  stat: string;            // 属性名，如 "life", "fire_resistance", "attack_speed"
  modType: "flat" | "percent" | "increased" | "more";
  min: number;
  max: number;
  rolled?: number;         // 实际roll出的值
}
```

## 3. 通货（Currency）

```typescript
interface Currency {
  id: string;
  name: string;
  description: string;
  rarity: "common" | "uncommon" | "rare";
  
  // 效果
  effect: CurrencyEffect;
}

// 通货效果（不同通货有不同类型的效果）
type CurrencyEffect =
  | { type: "upgrade_rarity"; from: Rarity; to: Rarity }
  | { type: "reforge_rare" }                    // Chaos: 重掷稀有装备全部词缀
  | { type: "reforge_magic" }                   // Alteration: 重掷魔法装备
  | { type: "add_prefix" }                      // Augmentation: 添加一条前缀
  | { type: "add_suffix" }                      // Augmentation: 添加一条后缀
  | { type: "annul" }                           // 宝珠：随机移除一条词缀
  | { type: "exalt" }                           // 崇高：添加一条高阶词缀
  | { type: "scour" }                           // 通货：清除所有词缀变白
  | { type: "quality"; amount: number }         // 磨刀石/护甲片
  | { type: "socket"; sockets: number }         // 工匠石：改变孔数
  | { type: "link" }                            // 链接石：随机连接孔
  | { type: "color"; color?: string }           // 聚变石：改变孔颜色
  | { type: "divine" }                          // 神圣：重roll数值不改词缀
  | { type: "fracture" };                       // 锁定：锁定一条词缀后重掷
```

## 4. 宝石（Gem）

```typescript
enum GemColor { Red = "red", Green = "green", Blue = "blue" }

// 宝石类型
enum GemType {
  Active = "active",       // 主动技能宝石
  Support = "support"      // 辅助宝石
}

interface Gem {
  id: string;
  name: string;
  type: GemType;
  color: GemColor;
  level: number;           // 宝石等级 1-20
  requiredLevel: number;   // 角色等级要求
  
  // 主动宝石
  skill?: ActiveSkill;
  
  // 辅助宝石
  support?: SupportEffect;
}

// 主动技能
interface ActiveSkill {
  baseDamage: StatBonus[];
  tags: string[];          // 用于匹配辅助宝石
  castTime: number;        // 施法时间（回合制中为行动点数消耗）
  manaCost: number;
  damageType: DamageType;
  description: string;
}

// 辅助效果
interface SupportEffect {
  // 放大倍率
  multiplier: number;       // 如 1.3 = +30% 伤害
  addedTags: string[];      // 给被辅助的技能增加标签
  specialEffect?: string;   // 特殊效果标识（如 CastOnCrit）
  excludesTags: string[];   // 不能同时存在的标签（防冲突）
}
```

## 5. 技能组（Skill Link）

```typescript
interface SkillGroup {
  id: string;
  name: string;             // 玩家自定义
  activeGem: Gem;           // 主动宝石（1个）
  supportGems: Gem[];       // 辅助宝石（0-5个，受装备孔数限制）
  
  // 计算后的最终技能数据
  computed?: ComputedSkill;
}

interface ComputedSkill {
  totalDamage: number;
  damageType: DamageType;
  tags: string[];           // 经辅助宝石修改后的最终标签
  manaCost: number;
  castTime: number;
  specialEffects: string[];
}
```

## 6. 天赋树（Passive Tree）

```typescript
interface PassiveNode {
  id: string;
  name: string;
  type: "normal" | "notable" | "keystone" | "ascendancy";
  x: number;               // 坐标（用于渲染）
  y: number;
  connections: string[];    // 连接的节点ID
  stats: StatBonus[];       // 节点提供的属性
  requires: string[];       // 前置节点
  allocated: boolean;       // 是否已点亮
}

interface PassiveTree {
  nodes: Map<string, PassiveNode>;
  // 按区域分组方便查询
  clusters: PassiveCluster[];
}

interface PassiveCluster {
  name: string;             // "血量圈" "暴击圈" 等
  center: { x: number; y: number };
  nodes: string[];          // 包含的节点ID
}
```

## 7. 角色（Player）

```typescript
interface Player {
  name: string;
  level: number;
  experience: number;
  
  // 基础属性（受天赋/装备影响）
  baseStats: {
    strength: number;       // +力量 = +生命, +物理伤害
    dexterity: number;      // +敏捷 = +命中, +闪避
    intelligence: number;   // +智力 = +能量护盾, +法力
  };
  
  // 资源
  life: number;
  maxLife: number;
  mana: number;
  maxMana: number;
  manaReserved: number;     // 光环保留
  energyShield: number;     // 能量护盾
  
  // 防御（由装备/天赋计算）
  defenses: {
    armor: number;
    evasion: number;
    energyShield: number;
    fireRes: number;        // 抗性（上限75）
    coldRes: number;
    lightningRes: number;
    chaosRes: number;
    blockChance: number;
  };
  
  // 进攻（由装备/天赋/技能计算）
  offense: {
    increasedDamage: number;
    moreDamage: number;     // 乘算
    attackSpeed: number;
    critChance: number;
    critMultiplier: number;
    accuracy: number;
  };
  
  // 天赋点
  passivePoints: number;
  allocatedNodes: string[];
  
  // 升华
  ascendancy?: AscendancyChoice;
  
  // 装备
  equipment: Partial<Record<EquipSlot, Item>>;
  
  // 技能组
  skillGroups: SkillGroup[];
  
  // 背包
  inventory: Inventory;
}

interface Inventory {
  items: Item[];
  currencies: Map<string, number>;  // currencyId -> 数量
  maxSlots: number;                 // 可扩展
}
```

## 8. 伤害类型

```typescript
enum DamageType {
  Physical = "physical",
  Fire = "fire",
  Cold = "cold",
  Lightning = "lightning",
  Chaos = "chaos"
}

enum AilmentType {
  Ignite = "ignite",       // 点燃 - 持续火焰伤害
  Freeze = "freeze",       // 冻结 - 不能行动
  Shock = "shock",         // 感电 - 受到伤害增加
  Bleed = "bleed",         // 流血 - 持续物理伤害
  Poison = "poison",       // 中毒 - 持续混沌伤害
  Impale = "impale"        // 穿刺 - 存储物理伤害
}
```
